const int numToggles = 20;

// Megaboard pin order:
// 53 is 1st button, 51 is 2nd button, 49 is 3rd button, etc.
// The ESP32 is NOT directly driving these pins.
// Commands (e.g. 1:ON, 2:OFF) are sent through Serial1 to Arduino Mega.
const int ledPins[] = {
  53, 51, 49, 47, 45, 43, 41, 39, 37, 35,
  33, 31, 52, 50, 48, 46, 44, 42, 40, 38
};

// ============================================================
// HIDDEN LAMPS (Empty - all 20 buttons visible)
// ============================================================

const int hiddenToggles[] = {};

const int hiddenCount =
  sizeof(hiddenToggles) / sizeof(hiddenToggles[0]);

// ============================================================
// OLED
// ============================================================

#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1

  SCREEN_WIDTH,
  SCREEN_HEIGHT,
  &Wire,
  OLED_RESET
);

// ============================================================
// FUNCTION DECLARATIONS
// ============================================================

String generateHTML();

void handleToggle();

bool isHiddenToggle(int toggleNumber);

int pinForToggle(int toggleNumber);

// ============================================================
// CHECK WHETHER A LAMP IS HIDDEN
// ============================================================

bool isHiddenToggle(int toggleNumber) {

  for (int i = 0; i < hiddenCount; i++) {

    if (hiddenToggles[i] == toggleNumber) {
      return true;
    }
  }

  return false;
}

// ============================================================
// GET PIN NUMBER FOR TOGGLE
// ============================================================

int pinForToggle(int toggleNumber) {

  if (toggleNumber < 1 || toggleNumber > numToggles) {
    return -1;
  }

  return ledPins[toggleNumber - 1];
}

// ============================================================
// HANDLE LAMP TOGGLE
// ============================================================

void handleToggle() {

  // Check that both parameters exist
  if (!server.hasArg("state") ||
      !server.hasArg("toggle")) {

    server.send(
      400,
      "text/plain",
      "Bad Request"
    );

    return;
  }

  // Read values
  int toggleNumber =
    server.arg("toggle").toInt();

  int state =
    server.arg("state").toInt();

  // Validate toggle number
  if (toggleNumber < 1 ||
      toggleNumber > numToggles) {

    server.send(
      400,
      "text/plain",
      "Invalid Toggle Number"
    );

    return;
  }

  // Only allow 0 or 1
  if (state != 0 && state != 1) {

    server.send(
      400,
      "text/plain",
      "Invalid State"
    );

    return;
  }

  // Convert state to text
  String toggleStateStr =
    (state == 1) ? "ON" : "OFF";

  // ========================================================
  // SEND COMMAND TO YOUR EXTERNAL CONTROLLER
  // ========================================================
  //
  // Format:
  //
  //     1:ON
  //     1:OFF
  //
  // ========================================================

  String commandStr = String(toggleNumber) + ":" + toggleStateStr;
  Serial1.println(commandStr);

  Serial.print(F("[ESP32] Button "));
  Serial.print(toggleNumber);
  Serial.print(F(" (Mega Pin "));
  Serial.print(pinForToggle(toggleNumber));
  Serial.print(F(") -> "));
  Serial.print(toggleStateStr);
  Serial.print(F(" | Sent to Mega: "));
  Serial.println(commandStr);

  // Send response to browser
  server.send(
    200,
    "text/plain",
    "OK"
  );
}

function sendToggle(toggleNumber) {

    var button =
        document.getElementById(
            "button" + toggleNumber
        );

    // Safety check
    if (!button) {
        return;
    }

    // Read current state
    var currentState =
        button.getAttribute(
            "data-state"
        );

    // Change state
    var state =
        currentState === "ON"
            ? 0
            : 1;

    // Disable button temporarily
    button.disabled = true;

    // Create HTTP request
    var xhr =
        new XMLHttpRequest();

    xhr.open(
        "GET",
        "/toggle?state=" +
        state +
        "&toggle=" +
        toggleNumber,
        true
    );

    xhr.onreadystatechange =
        function() {

            if (
                xhr.readyState === 4
            ) {

                // Re-enable button
                button.disabled = false;

                // Server accepted command
                if (
                    xhr.status === 200
                ) {

                    var newState =
                        state === 1
                            ? "ON"
                            : "OFF";

                    button.setAttribute(
                        "data-state",
                        newState
                    );

                    var label =
                        button.getAttribute(
                            "data-label"
                        );

                    button.innerHTML =
                        label +
                        "<br><small>" +
                        newState +
                        "</small>";

                }

            }

        };

    // Send request
    xhr.send();

}

// SETUP

void setup() {

  // ----------------------------------------------------------
  // SERIAL MONITOR
  // ----------------------------------------------------------

  Serial.begin(115200);

  delay(500);

  Serial.println();
  Serial.println(
    "======================================"
  );

  Serial.println(
    "YATHRA CONTROL SYSTEM"
  );

  Serial.println(
    "======================================"
  );

  // ----------------------------------------------------------
  // SERIAL1
  // ----------------------------------------------------------
  //
  // RX = GPIO 18
  // TX = GPIO 19
  //
  // This is kept exactly according to your original code.
  //
  // ----------------------------------------------------------

  Serial1.begin(
    9600,
    SERIAL_8N1,
    18,
    19
  );

  // ----------------------------------------------------------
  // I2C
  // ----------------------------------------------------------
  //
  // SDA = GPIO 21
  // SCL = GPIO 22
  //
  // ----------------------------------------------------------

  Wire.begin(21, 22);

  // ----------------------------------------------------------
  // OLED INITIALIZATION
  // ----------------------------------------------------------

  if (
    display.begin(
      SSD1306_SWITCHCAPVCC,
      0x3C
    )
  ) {

    display.clearDisplay();

    display.setTextColor(
      SSD1306_WHITE
    );

    display.setTextSize(1);

    display.setCursor(0, 0);

    display.println(
      "YATHRA 2026"
    );

    display.println();

    display.println(
      "Starting WiFi..."
    );

    display.display();

  } else {

    Serial.println(
      "OLED not detected."
    );

  }

  // ----------------------------------------------------------
  // WIFI MODE
  // ----------------------------------------------------------

  WiFi.mode(WIFI_STA);

  WiFi.begin(
    ssid,
    password
  );

  Serial.print(
    "Connecting to WiFi"
  );

  // ----------------------------------------------------------
  // WAIT FOR WIFI
  // ----------------------------------------------------------

  int attempts = 0;

  while (
    WiFi.status() != WL_CONNECTED &&
    attempts < 60
  ) {

    delay(500);

    Serial.print(".");

    attempts++;

  }

  Serial.println();

  // ----------------------------------------------------------
  // WIFI RESULT
  // ----------------------------------------------------------

  if (
    WiFi.status() == WL_CONNECTED
  ) {

    Serial.println(
      "WiFi connected."
    );

    Serial.print(
      "SSID: "
    );

    Serial.println(
      ssid
    );

    Serial.print(
      "IP Address: "
    );

    Serial.println(
      WiFi.localIP()
    );

    // OLED
    display.clearDisplay();

    display.setTextSize(1);

    display.setCursor(0, 0);

    display.println(
      "SYSTEM ONLINE"
    );

    display.println();

    display.print(
      "IP:"
    );

    display.println(
      WiFi.localIP()
    );

    display.display();

  } else {

    Serial.println(
      "WiFi connection failed."
    );

    // OLED
    display.clearDisplay();

    display.setTextSize(1);

    display.setCursor(0, 0);

    display.println(
      "WIFI ERROR"
    );

    display.println();

    display.println(
      "Check SSID/PASSWORD"
    );

    display.display();

  }

  // ----------------------------------------------------------
  // MAIN WEB PAGE
  // ----------------------------------------------------------

  server.on(
    "/",
    HTTP_GET,
    []() {

      server.send(
        200,
        "text/html",
        generateHTML()
      );

    }
  );

  // ----------------------------------------------------------
  // TOGGLE API
  // ----------------------------------------------------------

  server.on(
    "/toggle",
    HTTP_GET,
    []() {

      handleToggle();

    }
  );

  // ----------------------------------------------------------
  // 404 HANDLER
  // ----------------------------------------------------------

  server.onNotFound(
    []() {

      server.send(
        404,
        "text/plain",
        "404 - Page Not Found"
      );

    }
  );

  // ----------------------------------------------------------
  // START SERVER
  // ----------------------------------------------------------

  server.begin();

  Serial.println(
    "HTTP server started."
  );

  Serial.println();

  if (
    WiFi.status() == WL_CONNECTED
  ) {

    Serial.print(
      "Open this address in browser: "
    );

    Serial.print(
      "http://"
    );

    Serial.println(
      WiFi.localIP()
    );

  }

  Serial.println(
    "======================================"
  );
}

// ============================================================
// LOOP
// ============================================================

void loop() {

  // Handle web requests
  server.handleClient();

  // Small delay allows other ESP32 tasks to run
  delay(2);
}
